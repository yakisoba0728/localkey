#!/bin/busybox sh
export PATH=/bin:/sbin:/usr/bin:/usr/sbin
export PYTHONUNBUFFERED=1
mount -t proc proc /proc
mount -t sysfs sysfs /sys
mount -t devtmpfs devtmpfs /dev
mount -t tmpfs tmpfs /tmp
mkdir -p /dev/pts /guest-state /dev/gadget
mount -t devpts devpts /dev/pts
ip link set lo up
ske_probe_rc=80
if /usr/lib/systemd/systemd-udevd --daemon --resolve-names=never && /sbin/modprobe hid-generic && /sbin/modprobe udc-core && /sbin/insmod /opt/dummy_hcd.ko is_high_speed=0 requested_ep_addresses=1 && /sbin/modprobe gadgetfs && /sbin/modprobe usbhid && mount -t gadgetfs gadgetfs /dev/gadget; then
    /sbin/modprobe libcomposite && /sbin/modprobe usb_f_fs && mount -t configfs configfs /sys/kernel/config
    mkdir -p /sys/kernel/config/usb_gadget/ske/strings/0x409 /sys/kernel/config/usb_gadget/ske/configs/c.1 /sys/kernel/config/usb_gadget/ske/functions/ffs.ske /dev/ffs-ske
    ln -s /sys/kernel/config/usb_gadget/ske/functions/ffs.ske /sys/kernel/config/usb_gadget/ske/configs/c.1/ske
    mount -t functionfs ske /dev/ffs-ske
    /opt/ske-venv/bin/python /opt/verification/verify-usb-large-partial-runtime-prototype.py
    ske_probe_rc=$?
fi
echo SKE_USB_LARGE_PARTIAL_RUNTIME_PROBE_GUEST_EXIT=$ske_probe_rc
sync
poweroff -f
