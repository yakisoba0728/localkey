"""Run unmodified frozen40 large partial runtime experiment in a fresh owned guest."""
import hashlib,json,subprocess
from pathlib import Path
probe=Path('work/linux-usb-large-partial-runtime-probe').resolve();project=probe/'frozen40/security-key-emulator';helpers=probe/'helpers'
artifacts=Path('work/linux-usb-large-partial-runtime-probe-artifacts').resolve();artifacts.mkdir(exist_ok=True)
build=project/'bin/linux-arm64'
archive=Path('work/security-key-emulator-v40.zip')
with archive.open('rb') as f:archive_sha=hashlib.file_digest(f,'sha256').hexdigest()
assert archive_sha=='c327a32eb64f6f8fdb1fc0a343cf76e8c52d5680e170ae623f77f2a79ea6a7d5'
subprocess.run(['docker','run','--rm','--network','none','--platform','linux/arm64',
 '--mount',f'type=bind,source={artifacts},target=/artifacts',
 '--mount',f'type=bind,source={build},target=/artifacts/build,readonly',
 '--mount',f'type=bind,source={helpers},target=/work,readonly',
 '--mount',f'type=bind,source={project}/tools,target=/src-tools,readonly',
 '--mount',f'type=bind,source={project},target=/src-application,readonly',
 'ske-linux-native:0.36','python3','/work/build-guest.py'],check=True)
command=['docker','run','--rm','--network','none','--name','ske-linux-usb-large-partial-runtime-probe',
 '--mount',f'type=bind,source={artifacts},target=/artifacts,readonly',
 'ske-linux-native:0.36','timeout','--kill-after=10s','300s','qemu-system-aarch64',
 '-machine','virt','-cpu','cortex-a72','-accel','tcg','-m','2048','-smp','2',
 '-kernel','/artifacts/guest-kernel','-initrd','/artifacts/guest-initramfs.gz',
 '-append','console=ttyAMA0 rdinit=/init panic=1','-nic','none','-display','none','-serial','stdio','-monitor','none','-no-reboot']
console=artifacts/'console.log'
with console.open('wb') as log:runner=subprocess.run(command,stdout=log,stderr=subprocess.STDOUT)
s=console.read_text(errors='replace');report={'passed':False,'runner_exit':runner.returncode}
try:
 a,b='SKE_USB_LARGE_PARTIAL_RUNTIME_PROBE_BEGIN','SKE_USB_LARGE_PARTIAL_RUNTIME_PROBE_END';assert s.count(a)==s.count(b)==1
 report=json.loads(s.split(a,1)[1].split(b,1)[0].strip());report['runner_exit']=runner.returncode
 for name,key in [('swpasskeyd','emulator_binary_sha256'),('ske-usbd','usb_bridge_binary_sha256')]:assert report[key]==hashlib.file_digest((build/name).open('rb'),'sha256').hexdigest()
 assert report['helper_sha256']==hashlib.file_digest((helpers/'verify-usb-large-partial-runtime-prototype.py').open('rb'),'sha256').hexdigest()
 assert report['production_source_modified_for_this_original_trial'] is False and report['original_frozen40_source_and_binaries'] is True
 assert runner.returncode==0 and 'SKE_USB_LARGE_PARTIAL_RUNTIME_PROBE_GUEST_EXIT=0' in s
 assert report['passed'],report.get('error')
 assert len(report['cycles'])==6 and len(report['assertions'])==7
 assert [c['total_response_bytes'] for c in report['cycles']]==[2047,2048,2049,3071,3072,3073]
 for c in report['cycles']:
  assert c['exact_native_original_tail_verified'] and c['wait_for_resume_returns']==[0]
  assert c['before_usb']==c['after_native_tail_usb']
except Exception as exc:report.update(passed=False,host_gate_error=f'{type(exc).__name__}: {exc}')
report['original_frozen_archive_sha256']=archive_sha
report['original_linux_source_sha256']=hashlib.file_digest((project/'source/src/usb/linux.cpp').open('rb'),'sha256').hexdigest()
report['prototype_helper_host_sha256']=hashlib.file_digest((helpers/'verify-usb-large-partial-runtime-prototype.py').open('rb'),'sha256').hexdigest()
report['console_sha256']=hashlib.file_digest(console.open('rb'),'sha256').hexdigest()
(artifacts/'result.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({key:report.get(key) for key in ['passed','test_count','runner_exit','error','cleanup_error','host_gate_error','original_frozen_archive_sha256']},indent=2))
raise SystemExit(0 if report['passed'] else 1)
