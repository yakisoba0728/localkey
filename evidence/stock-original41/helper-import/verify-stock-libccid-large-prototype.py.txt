class StockFixture(base.CompositeFixture):
    def __init__(self,directory):
        super().__init__(Path('/opt/swpasskeyd'),directory)
        self.connection=self.manager=None
    def start(self):
        super().start()
        empty=self.directory/'empty-readers';empty.mkdir()
        self.manager_log=(self.directory/'pcscd.log').open('ab')
        self.manager=subprocess.Popen(['/usr/sbin/pcscd','--foreground','--debug','--apdu','--disable-polkit','--config',str(empty)],
            stdout=self.manager_log,stderr=self.manager_log)
        return self.connect_own()
    def connect_own(self):
        def discover():
            if self.manager.poll() is not None:raise RuntimeError((self.directory/'pcscd.log').read_text())
            try:
                found=list(readers())
                return next((r for r in found if self.serial in r.name),None)
            except (EstablishContextException,ListReadersException):return None
        reader=wait(discover,'stock private pcscd finds only own serial USB reader',12)
        self.reader_name=reader.name
        self.connection=ScardSmartCardConnection(reader.createConnection())
        assert self.connection.transport==TRANSPORT.USB
        assert self.connection.connection.getATR()==list(bytes.fromhex('3bf011000001e0'))
        log=(self.directory/'pcscd.log').read_text()
        assert 'libccid.so' in log and 'ske-ifd.so' not in log
        return self.connection
    def stop(self):
        if self.connection:self.connection.close();self.connection=None
        if self.manager:
            self.manager.terminate();self.manager.wait(10);self.manager=None
        if hasattr(self,'manager_log'):self.manager_log.close()
        super().stop()


def main():
    f=tmp=trace=None
    report={'passed':False,'tests':checks,'cycles':[],'assertions':[],'piv_signatures':[],
        'scope':'actual original41 profile6 USB -> unmodified stock libccid -> private pcscd -> stock PCSC/Yubico SDK large transfers',
        'physical_device_access':False,'host_device_passthrough':False,'host_hid_drivers_unbound':False,
        'native_macos_registration_verified':False,'production_udc_verified':False,'physical_touch_verified':False,
        'manufacturer_trust_reproduced':False,'card_testing_auto_allow':False,
        'fido_testing_auto_allow_only_for_owned_fixture':True,'own_native_ccid_driver_used':False,
        'sdk_or_stock_driver_binary_patched':False,'production_source_modified_for_this_original_trial':False,
        'opaque_piv_object_is_valid_chuid':False,'stock_piv_oath_sdk_unmodified':True}
    try:
        check(Path('/opt/isolated-ske-usb-guest').exists() and platform.release()=='6.8.0-146-generic'
              and platform.machine()=='aarch64' and [p.name for p in Path('/sys/class/net').iterdir()]==['lo'],
              'actual own noNIC ARM64 distro guest before any device action')
        # Only the documented COMM logging bit changes, inside this disposable guest.
        info=Path('/usr/lib/pcsc/drivers/ifd-ccid.bundle/Contents/Info.plist')
        before=info.read_bytes();old=plistlib.loads(before)
        updated=re.sub(rb'(<key>ifdLogLevel</key>\s*<string>)[^<]+(</string>)',rb'\g<1>0x0007\2',before,count=1)
        after=plistlib.loads(updated)
        expected=copy.deepcopy(old);expected['ifdLogLevel']='0x0007'
        check(after==expected,'stock driver Info.plist only documented CRITICAL INFO COMM log mask changed')
        info.write_bytes(updated)
        driver=Path('/usr/lib/pcsc/drivers/ifd-ccid.bundle/Contents/Linux/libccid.so')
        report['driver_binary_sha256_before']=hashlib.sha256(driver.read_bytes()).hexdigest()
        report['driver_info_plist_logging_only']={'before_sha256':hashlib.sha256(before).hexdigest(),
            'after_sha256':hashlib.sha256(updated).hexdigest(),'before_value':old['ifdLogLevel'],'after_value':after['ifdLogLevel'],
            'all_other_parsed_fields_unchanged':True}
        tmp=Path(tempfile.mkdtemp(prefix='own-stock-large-',dir='/guest-state'))
        os.environ['SKE_TRACE_USB']='1'
        f=StockFixture(tmp);connection=f.start()
        report['service_pids']=[f.bridge.pid,f.process.pid,f.card.process.pid,f.manager.pid]
        report['controller']=f.controller
        report['initial_reader_name']=f.reader_name
        protocol=PinProtocolV2();ct=f.ctap
        ClientPin(ct,protocol).set_pin(state.PIN)
        token=ClientPin(ct,protocol).get_pin_token(state.PIN,ClientPin.PERMISSION.MAKE_CREDENTIAL,state.RP)
        challenge=hashlib.sha256(b'own stock-PCSC large initial registration').digest()
        created=ct.make_credential(challenge,{'id':state.RP},{'id':state.USER,'name':'own-stock-large'},
            [{'type':'public-key','alg':-7}],options={'rk':True},pin_uv_protocol=2,
            pin_uv_param=protocol.authenticate(token,challenge))
        credential=created.auth_data.credential_data
        check(created.auth_data.flags&5==5,'original native resident FIDO key created with PIN UVp2')
        state.close_hid(f);shared.prove_fido(f,credential,'initial',report)
        original=state.reboot.snapshot(f.serial);report['original_identity']=original
        f.connection.close();f.connection=None
        with state.reboot.connection('FIDO',f.serial) as management:
            ManagementSession(management).write_device_config(DeviceConfig({TRANSPORT.USB:CAPABILITY(0x33a)}),reboot=True)
        identity=wait(lambda:(lambda x:x if x and x['pid']==0x406 and x['address']!=original['address'] else None)(state.reboot.snapshot(f.serial)),
                      'actual original profile6 native reenumeration',10)
        report['profile6_identity']=identity
        saved_reference=json.loads(Path('/opt/usb-reference-observation.json').read_text())
        check(identity['host_visible_configuration_descriptor_hex']==saved_reference['configuration_descriptor_hex'],
              'stock reader profile6 exact preserved reference configuration bytes')
        connection=f.connect_own();report['reader_name']=f.reader_name
        check(type(connection).__name__=='ScardSmartCardConnection' and connection.transport==TRANSPORT.USB,
              'actual stock pyscard/ykman USB connection; no own CCID driver or direct IFD')
        piv=PivSession(connection);piv.authenticate(DEFAULT_MANAGEMENT_KEY)
        original_key=piv.generate_key(SLOT.AUTHENTICATION,KEY_TYPE.ECCP256)
        der=original_key.public_bytes(serialization.Encoding.DER,serialization.PublicFormat.SubjectPublicKeyInfo)
        report['original_piv_public_key_spki_hex']=der.hex();report['original_piv_public_key_sha256']=hashlib.sha256(der).hexdigest()
        oath=OathSession(connection)
        own=oath.put_credential(CredentialData.parse_uri('otpauth://hotp/StockLarge:own?secret=JBSWY3DPEHPK3PXP&counter=0'))
        counter=lambda:EncryptedState(f.directory/'card/state.bin').data['oath']['credentials'][own.id.hex()]['counter']
        def durable():
            data=copy.deepcopy(EncryptedState(f.directory/'card/state.bin').data)
            data.pop('management',None);data['otp'].pop('sequence',None)
            return hashlib.sha256(json.dumps(data,sort_keys=True).encode()).hexdigest()
        trace=Path('/sys/kernel/tracing');runtime.ensure_mount('tracefs',trace)
        (trace/'buffer_size_kb').write_text('2048\n')
        names=['sys_enter_write','sys_exit_write','sys_enter_io_submit','sys_exit_io_submit','sys_enter_io_getevents','sys_exit_io_getevents']
        events=[trace/'events/syscalls'/name for name in names]
        report['trace_formats']={e.name:(e/'format').read_text() for e in events}
        def arm():
            tids=[int(p.name) for p in (Path('/proc')/str(f.bridge.pid)/'task').iterdir()]
            selected=' || '.join('common_pid == '+str(tid) for tid in tids)
            for event in events:
                (event/'filter').write_text(selected+'\n');(event/'enable').write_text('1\n')
            (trace/'trace').write_text('');(trace/'tracing_on').write_text('1\n')
        check(counter()==0,'original stock PCSC HOTP counter zero')
        for index,total in enumerate([2047,2048,2049,3071,3072,3073,4095,4096]):
            phase=str(total)+'-stock-libccid'
            opaque=bytes((offset*29+total)&255 for offset in range(total-16))
            expected_apdu=bytes(Tlv(0x53,opaque))+b'\x90\x00'
            check(len(expected_apdu)+10==total,'encoded intended stock CCID total response boundary '+phase)
            piv=PivSession(connection)
            if total>=3071:
                piv.protocol=SmartCardProtocol(connection);piv.protocol.configure(piv.version,force_short=True)
            piv.authenticate(DEFAULT_MANAGEMENT_KEY)
            before_log=(f.directory/'pcscd.log').stat().st_size
            arm()
            row={'index':index,'phase':phase,'total_response_bytes':total,'opaque_bytes':len(opaque),
                'opaque_sha256':hashlib.sha256(opaque).hexdigest(),'expected_get_apdu_hex':expected_apdu.hex(),
                'before_usb':state.reboot.snapshot(f.serial),'fresh_supported_short_processor':total>=3071}
            report['active_cycle']=row
            piv.put_object(OBJECT_ID.CHUID,opaque)
            row['after_put_plaintext_sha256']=durable()
            actual=piv.get_object(OBJECT_ID.CHUID)
            row['actual_sdk_object_hex']=actual.hex();row['actual_sdk_object_sha256']=hashlib.sha256(actual).hexdigest()
            row['after_get_plaintext_sha256']=durable();row['actual_kernel_trace']=(trace/'trace').read_text()
            time.sleep(.02)
            row['stock_pcscd_log']=(f.directory/'pcscd.log').read_bytes()[before_log:].decode(errors='replace')
            row['usb_log_after_get']=(f.directory/'usb.log').read_text(errors='replace')[-18000:]
            check(actual==opaque,'actual unmodified PIV SDK exact original object through stock libccid '+phase)
            check(row['after_put_plaintext_sha256']==row['after_get_plaintext_sha256'] and counter()==index,
                  'stock SDK large GET leaves own plaintext/HOTP unchanged '+phase)
            piv.verify_pin('123456')
            digest=hashlib.sha256(('own stock large original PIV '+phase).encode()).digest()
            signature=piv.sign(SLOT.AUTHENTICATION,KEY_TYPE.ECCP256,digest,hashes.SHA256())
            original_key.verify(signature,digest,ec.ECDSA(hashes.SHA256()))
            report['piv_signatures'].append({'phase':phase,'digest_hex':digest.hex(),'signature_hex':signature.hex(),'original_key_signature_verified':True})
            check(True,'independent original PIV key signature after stock large transfer '+phase)
            value=OathSession(connection).calculate_code(own).value
            mac=hmac.new(base64.b32decode('JBSWY3DPEHPK3PXP'),index.to_bytes(8,'big'),'sha1').digest();offset=mac[-1]&15
            expected=str((int.from_bytes(mac[offset:offset+4],'big')&0x7fffffff)%1000000).zfill(6)
            check(value==expected and counter()==index+1,'independent stock native HOTP response and single commit '+phase)
            shared.prove_fido(f,credential,phase,report)
            row['after_usb']=state.reboot.snapshot(f.serial)
            check(row['after_usb']==identity and all(p.poll() is None for p in [f.bridge,f.process,f.card.process,f.manager]),
                  'same USB generation and four original services after stock large transfer '+phase)
            report['cycles'].append(row);report.pop('active_cycle')
        report['driver_binary_sha256_after']=hashlib.sha256(driver.read_bytes()).hexdigest()
        check(report['driver_binary_sha256_before']==report['driver_binary_sha256_after'],'stock libccid binary never modified')
        report['final_hotp_counter']=counter();report['passed']=True
    except Exception as exc:
        report.update(error=f'{type(exc).__name__}: {exc}',traceback=traceback.format_exc())
        if trace and (trace/'trace').exists():report['last_actual_kernel_trace']=(trace/'trace').read_text()
    finally:
        if trace:
            for name in ['sys_enter_write','sys_exit_write','sys_enter_io_submit','sys_exit_io_submit','sys_enter_io_getevents','sys_exit_io_getevents']:
                event=trace/'events/syscalls'/name/'enable'
                if event.exists():event.write_text('0\n')
            if (trace/'tracing_on').exists():(trace/'tracing_on').write_text('0\n')
        if f:
            for name in ['pcscd.log','usb.log','daemon.log','card/log.txt']:
                path=f.directory/name
                if path.exists():report[name]=path.read_text(errors='replace')
            report['final_service_status']={name:p.poll() for name,p in [('bridge',f.bridge),('fido',f.process),('card',f.card.process),('pcscd',f.manager)] if p is not None}
            try:f.stop()
            except Exception as exc:report.update(passed=False,cleanup_error=f'{type(exc).__name__}: {exc}')
        if tmp and tmp.exists():shutil.rmtree(tmp)
        report['test_count']=len(checks)
        for name,key in [('swpasskeyd','emulator_binary_sha256'),('ske-usbd','usb_bridge_binary_sha256')]:report[key]=hashlib.sha256(Path('/opt',name).read_bytes()).hexdigest()
        report['card_service_sha256']=hashlib.sha256(Path('/opt/emucard/service.py').read_bytes()).hexdigest()
        report['helper_sha256']=hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
        report['shared_fido_helper_sha256']=hashlib.sha256(Path(shared.__file__).read_bytes()).hexdigest()
        print('SKE_STOCK_LIBCCID_LARGE_PROBE_BEGIN\n'+json.dumps(report)+'\nSKE_STOCK_LIBCCID_LARGE_PROBE_END',flush=True)
    return 0 if report['passed'] else 1

if __name__=='__main__':raise SystemExit(main())
