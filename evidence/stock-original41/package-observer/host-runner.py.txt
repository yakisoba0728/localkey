"""Run original frozen41 stock CCID large experiment; no host device/network."""
import hashlib,json,subprocess
from pathlib import Path
probe=Path('work/linux-stock-libccid-large-probe').resolve();project=probe/'frozen41/security-key-emulator';helpers=probe/'helpers';build=project/'bin/linux-arm64'
artifacts=Path('work/linux-stock-libccid-large-probe-artifacts').resolve();artifacts.mkdir(exist_ok=True)
archive=Path('work/security-key-emulator-v41.zip')
with archive.open('rb') as f:archive_sha=hashlib.file_digest(f,'sha256').hexdigest()
assert archive_sha=='2af3b19f3ae804b974968eec176e00d26df32c28a6da748246853145efa55c91'
subprocess.run(['docker','run','--rm','--network','none','--platform','linux/arm64',
 '--mount',f'type=bind,source={artifacts},target=/artifacts','--mount',f'type=bind,source={build},target=/artifacts/build,readonly',
 '--mount',f'type=bind,source={helpers},target=/work,readonly','--mount',f'type=bind,source={project}/tools,target=/src-tools,readonly',
 '--mount',f'type=bind,source={project},target=/src-application,readonly','ske-linux-native:0.36','python3','/work/build-guest.py'],check=True)
command=['docker','run','--rm','--network','none','--name','ske-stock-libccid-large-probe',
 '--mount',f'type=bind,source={artifacts},target=/artifacts,readonly','ske-linux-native:0.36','timeout','--kill-after=10s','300s','qemu-system-aarch64',
 '-machine','virt','-cpu','cortex-a72','-accel','tcg','-m','4096','-smp','2','-kernel','/artifacts/guest-kernel','-initrd','/artifacts/guest-initramfs.gz',
 '-append','console=ttyAMA0 rdinit=/init panic=1','-nic','none','-display','none','-serial','stdio','-monitor','none','-no-reboot']
console=artifacts/'console.log'
with console.open('wb') as log:runner=subprocess.run(command,stdout=log,stderr=subprocess.STDOUT)
s=console.read_text(errors='replace');report={'passed':False,'runner_exit':runner.returncode}
try:
 a,b='SKE_STOCK_LIBCCID_LARGE_PROBE_BEGIN','SKE_STOCK_LIBCCID_LARGE_PROBE_END';assert s.count(a)==s.count(b)==1
 report=json.loads(s.split(a,1)[1].split(b,1)[0].strip());report['runner_exit']=runner.returncode
 for name,key in [('swpasskeyd','emulator_binary_sha256'),('ske-usbd','usb_bridge_binary_sha256')]:assert report[key]==hashlib.file_digest((build/name).open('rb'),'sha256').hexdigest()
 assert report['helper_sha256']==hashlib.file_digest((helpers/'verify-stock-libccid-large-prototype.py').open('rb'),'sha256').hexdigest()
 assert report['own_native_ccid_driver_used'] is False and report['production_source_modified_for_this_original_trial'] is False
 assert runner.returncode==0 and 'SKE_STOCK_LIBCCID_LARGE_PROBE_GUEST_EXIT=0' in s
 assert report['passed'],report.get('error')
 assert len(report['cycles'])==8 and len(report['assertions'])==9 and len(report['piv_signatures'])==8
 assert report['final_hotp_counter']==8
except Exception as exc:report.update(passed=False,host_gate_error=f'{type(exc).__name__}: {exc}')
report['original_frozen_archive_sha256']=archive_sha
report['original_linux_source_sha256']=hashlib.file_digest((project/'source/src/usb/linux.cpp').open('rb'),'sha256').hexdigest()
report['prototype_helper_host_sha256']=hashlib.file_digest((helpers/'verify-stock-libccid-large-prototype.py').open('rb'),'sha256').hexdigest()
report['console_sha256']=hashlib.file_digest(console.open('rb'),'sha256').hexdigest()
(artifacts/'result.json').write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({key:report.get(key) for key in ['passed','test_count','runner_exit','error','cleanup_error','host_gate_error','original_frozen_archive_sha256']},indent=2))
raise SystemExit(0 if report['passed'] else 1)
