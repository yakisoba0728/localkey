def member(name, payload, mode, ino):
    name = name.encode()+b'\0'
    fields = [ino,mode,0,0,1,0,len(payload),0,0,0,0,len(name),0]
    h = b'070701'+b''.join(f'{x:08x}'.encode() for x in fields)
    h += name
    h += b'\0'*((-len(h))%4)
    h += payload+b'\0'*((-len(payload))%4)
    return h
