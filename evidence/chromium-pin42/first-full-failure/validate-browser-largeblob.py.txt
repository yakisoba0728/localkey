"""Validate the bounded stock-browser report against independently joined data."""

def validate(report):
    assert report['passed'] and report['test_count'] == len(report['tests']) == 29
    assert all(item['passed'] for item in report['tests'])
    assert all(report[key] is True for key in [
        'browser_large_blob_write_verified', 'browser_large_blob_read_verified',
        'browser_large_blob_independent_sdk_verified', 'browser_large_blob_restart_verified',
        'browser_large_blob_live_usb_reboot_verified',
        'own_guest_accessibility_enabled_before_browser_launch',
        'stock_chromium_es384_parser_boundary_verified',
    ])
    assert all(report[key] is False for key in [
        'physical_device_access', 'host_usb_passthrough', 'virtual_authenticator_enabled',
        'webauthn_api_replaced', 'browser_webauthn_service_callbacks_invoked_directly',
        'physical_touch_verified', 'manufacturer_trust_reproduced',
        'all_browser_extensions_verified', 'all_browser_algorithms_verified',
        'native_macos_registration_verified',
    ])
    assert report['browser_algorithms_verified'] == [-7, -8]
    assert report['browser_algorithms_rejected'] == [-35]
    assert report['browser_version']['product'] == 'Chrome/154.0.8037.92'
    assert report['browser_version']['revision'] == '@334b65d254ccc35df4fca82706d1753227b01039'
    assert report['browser_binary_sha256'] == '4f8459f639cd128f701fc84b04bbdc496a56202cdd78beec5db721bdda260911'
    assert all(command in ['Browser.getVersion', 'Runtime.evaluate', 'Page.navigate']
               for command in report['cdp_commands'])
    registrations, assertions, references = (report[key] for key in
        ['registrations', 'assertions', 'independent_sdk_references'])
    assert len(registrations) == len(references) == 2 and len(assertions) == 6
    assert {row['algorithm'] for row in registrations} == {-7, -8}
    assert {row['algorithm'] for row in references} == {-7, -8}
    assert len({row['large_blob_key_sha256'] for row in references}) == 2
    assert len({row['challenge_sha256'] for row in assertions}) == 6
    assert all(row['signature_verified'] and row['user_verified'] and row['user_handle_verified']
               for row in assertions)
    for reference in references:
        assert all(reference[key] is True for key in [
            'independent_sdk_signature_verified', 'independent_sdk_user_verified',
            'independent_sdk_user_handle_verified', 'independent_sdk_browser_blob_match',
            'independent_sdk_replacement_verified', 'unrelated_key_has_no_blob',
        ])
        assert reference['independent_sdk_pin_uv_protocol'] == 2
        assert reference['browser_payload_bytes'] == 792 and reference['sdk_payload_bytes'] == 1056
        rows = [row for row in assertions if row['algorithm'] == reference['algorithm']]
        assert len(rows) == 3
        reads = [row for row in rows if row.get('independent_sdk_blob_match')]
        assert len(reads) == 2 and {row['phase'] for row in reads} == {
            'browser-restart-after-sdk-write', 'live-usb-reboot'}
        assert all(row['large_blob_payload_sha256'] == reference['sdk_payload_sha256']
                   and row['large_blob_payload_bytes'] == reference['sdk_payload_bytes'] for row in reads)
    rejection = report['es384_rejection']
    assert rejection['browser_error']['error'] == 'NotAllowedError'
    assert (rejection['cose_algorithm'], rejection['cose_curve'], rejection['coordinate_bytes']) == (-35, 2, 48)
    assert all(rejection[key] is True for key in ['independent_sdk_signature_verified',
        'independent_sdk_user_verified', 'independent_sdk_user_handle_verified'])
    assert rejection['pin_uv_protocol'] == 2
    reboot = report['usb_reboot']
    assert reboot['same_three_service_processes'] is True
    assert reboot['before']['address'] != reboot['after']['address']
    assert reboot['before']['serial'] == reboot['after']['serial']
    for key in ['initial_metrics', 'before_reboot_metrics', 'final_metrics']:
        assert report[key]['pin_fail'] == 0 and report[key]['pin_retries'] == 8
    metrics = report['final_metrics']
    assert metrics['make_cred_ok'] == metrics['store_creds'] == 3
    assert metrics['get_assert_ok'] >= 9 and metrics['remaining'] == 97
    passwords = [row for row in report['ui_actions'] if row['role'] == 'password text']
    assert len(passwords) == 9 and all(row['actual_character_count'] == 15 for row in passwords)
    assert all(row['method'] in ['actual-guest-x-key-events', 'actual-at-spi-editable-text']
               for row in passwords)
    assert {(row['account'], row['phase'], row['mode']) for row in passwords} == {
        (f'own-chromium-blob-{algorithm}', phase, mode)
        for algorithm in [-7, -8]
        for phase, mode in [('initial', 'register'), ('initial', 'login'),
                            ('browser-restart-after-sdk-write', 'login'), ('live-usb-reboot', 'login')]
    } | {('own-chromium-rejected-es384', 'es384-negative', 'register')}
