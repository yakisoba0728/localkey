class NativeLongCcid(pending.NativeCcid):
    """Owned native CCID driver; stock PIV/OATH APDU SDK remains unchanged."""
    def __init__(self, device):
        self.history = []
        super().__init__(device)

    def write(self, kind, payload=b'', level=0):
        self.sequence = (self.sequence + 1) & 255
        packet = struct.pack('<BIBBBBB', kind, len(payload), 0, self.sequence, 0, level, 0) + payload
        assert self.device.write(2, packet, timeout=2000) == len(packet)
        self.history.append({'direction': 'out', 'kind': kind, 'bytes': len(packet),
                             'sequence': self.sequence, 'level': level})
        return self.sequence

    def read(self, sequence):
        header, payload = super().read(sequence)
        self.history.append({'direction': 'in', 'header': list(header), 'bytes': len(payload) + 10})
        # A host request of exactly3072 bytes ends without consuming the separate
        # CCID bulk ZLP. Drain it before submitting the next native CCID request.
        if len(payload) + 10 == CCID_MESSAGE_LENGTH:
            terminator = bytes(self.device.read(0x82, 1, timeout=1500))
            assert terminator == b'', ('original full-size CCID ZLP', terminator.hex())
            self.history.append({'direction': 'in', 'bytes': 0, 'terminator': True})
        return header, payload

    def exchange(self, kind, payload=b''):
        pieces = [payload[i:i + CCID_PAYLOAD_LENGTH] for i in range(0, len(payload), CCID_PAYLOAD_LENGTH)] or [b'']
        for index, piece in enumerate(pieces):
            level = 0 if len(pieces) == 1 else (1 if index == 0 else 2 if index == len(pieces) - 1 else 3)
            assert kind == 0x6f or len(pieces) == 1
            sequence = self.write(kind, piece, level)
            while True:
                header, data = self.read(sequence)
                if header[4] & 0xc0 == 0x80:
                    continue
                assert header[4] & 0xc0 == 0 and header[5] == 0, (header, data.hex())
                break
            if index < len(pieces) - 1:
                assert header[6] == 0x10 and not data
        complete = data
        while kind == 0x6f and header[6] in (1, 3):
            sequence = self.write(kind, level=0x10)
            header, data = self.read(sequence)
            assert header[4] & 0xc0 == 0 and header[5] == 0 and header[6] in (2, 3), header
            complete += data
        return header, complete


def main():
    f = tmp = d = trace = None
    report = {'passed': False, 'tests': checks, 'cycles': [], 'assertions': [],
              'scope': 'actual own large/partial CCID profile6 -> native FIDO Management profile2 -> profile6',
              'physical_device_access': False, 'host_device_passthrough': False,
              'host_hid_drivers_unbound': False, 'native_macos_registration_verified': False,
              'manufacturer_trust_reproduced': False, 'production_udc_verified': False,
              'physical_touch_verified': False, 'physical_firmware_reboot_timing_verified': False,
              'card_testing_auto_allow': False, 'fido_testing_auto_allow_only_for_owned_fixture': True,
              'stock_libccid_transport_used_for_this_fixture': False,
              'stock_piv_oath_sdk_unmodified': True, 'owned_ccid_driver_handles_command_and_response_chaining': True,
              'opaque_piv_object_is_valid_chuid': False,
              'unread_reply_reboot_grace_is_private_model_not_physical_timing': True}
    try:
        check(Path('/opt/isolated-ske-usb-guest').exists() and platform.release() == '6.8.0-146-generic'
              and platform.machine() == 'aarch64' and [p.name for p in Path('/sys/class/net').iterdir()] == ['lo'],
              'actual private ARM64 kernel USB guest with only loopback')
        check(CCID_MESSAGE_LENGTH == 3072 and CCID_PAYLOAD_LENGTH == 3062, 'saved CCID3072 envelope and3062 payload model')
        tmp = Path(tempfile.mkdtemp(prefix='own-long-cutover-', dir='/guest-state'))
        os.environ['SKE_TRACE_USB'] = '1'
        f = base.CompositeFixture(Path('/opt/swpasskeyd'), tmp)
        f.card = pending.PromptCardFixture(Path('/opt'), tmp / 'card', startup_timeout=30)
        ct = f.start()
        report['service_pids'] = [f.bridge.pid, f.process.pid, f.card.process.pid]
        report['controller'] = f.controller
        protocol = PinProtocolV2()
        ClientPin(ct, protocol).set_pin(state.PIN)
        token = ClientPin(ct, protocol).get_pin_token(state.PIN, ClientPin.PERMISSION.MAKE_CREDENTIAL, state.RP)
        challenge = hashlib.sha256(b'own native long CCID cutover registration').digest()
        response = ct.make_credential(challenge, {'id': state.RP}, {'id': state.USER, 'name': 'own-long-cutover'},
            [{'type': 'public-key', 'alg': -7}], options={'rk': True}, pin_uv_protocol=2,
            pin_uv_param=protocol.authenticate(token, challenge))
        credential = response.auth_data.credential_data
        check(response.auth_data.flags & 5 == 5, 'original native resident FIDO key created with PIN UVp2')
        state.close_hid(f)
        shared.prove_fido(f, credential, 'initial', report)
        report['original_identity'] = state.reboot.snapshot(f.serial)
        with state.reboot.connection('FIDO', f.serial) as conn:
            ManagementSession(conn).write_device_config(DeviceConfig({TRANSPORT.USB: CAPABILITY(0x33a)}), reboot=True)

        def new_generation(before, pid):
            return wait(lambda: (lambda value: value if value and value['pid'] == pid and value['address'] != before['address'] else None)
                        (state.reboot.snapshot(f.serial)), 'actual fresh native USB generation ' + hex(pid), 8)

        identity = new_generation(report['original_identity'], 0x406)
        report['profile6_identity'] = identity
        reference = json.loads(Path('/opt/usb-reference-observation.json').read_text())
        check(identity['host_visible_configuration_descriptor_hex'] == reference['configuration_descriptor_hex'],
              'initial native profile6 byte equal saved public physical configuration')
        backend = usb.backend.libusb1.get_backend(find_library=lambda name: '/usr/lib/aarch64-linux-gnu/libusb-1.0.so.0')

        report['libusb_generation_visibility'] = []

        def device(expected_identity):
            observation = {'expected_address': expected_identity['address'], 'attempts': []}
            report['libusb_generation_visibility'].append(observation)

            def matching_generation():
                matches = list(usb.core.find(find_all=True, idVendor=0x1050, idProduct=0x406, backend=backend))
                visible, owned = [], []
                for candidate in matches:
                    row = {'address': candidate.address}
                    try:
                        row['serial_matches'] = candidate.serial_number == f.serial
                    except usb.core.USBError as exc:
                        row['serial_error'] = str(exc)
                    else:
                        if row['serial_matches'] and candidate.address == expected_identity['address']:
                            owned.append(candidate)
                    visible.append(row)
                observation['attempts'].append(visible)
                if len(owned) == 1:
                    observation['selected_address'] = owned[0].address
                    observation['selected_serial_matches'] = True
                    return owned[0]
                assert len(owned) < 2, 'multiple devices with same owned serial and native address'
                return None

            return wait(matching_generation, 'libusb visibility of exact owned new USB generation', 4)

        d = device(identity)
        connection = NativeLongCcid(d)
        oath = OathSession(connection)
        own = oath.put_credential(CredentialData.parse_uri('otpauth://hotp/LongCutover:owned?secret=JBSWY3DPEHPK3PXP&counter=0'))
        count = lambda: EncryptedState(f.directory / 'card/state.bin').data['oath']['credentials'][own.id.hex()]['counter']

        def durable():
            data = copy.deepcopy(EncryptedState(f.directory / 'card/state.bin').data)
            data.pop('management', None)
            data['otp'].pop('sequence', None)
            return hashlib.sha256(json.dumps(data, sort_keys=True).encode()).hexdigest()

        def alive(label):
            check(all(p.poll() is None for p in [f.bridge, f.process, f.card.process]), 'same three original services alive ' + label)

        check(count() == 0, 'fresh owned HOTP counter zero')
        check(bytes(d.read(0x83, 8, timeout=1500)) == b'\x50\x03', 'initial actual occupied slot notification')
        trace = Path('/sys/kernel/tracing')
        runtime.ensure_mount('tracefs', trace)

        def arm():
            tids = [int(p.name) for p in (Path('/proc') / str(f.bridge.pid) / 'task').iterdir()]
            own_filter = ' || '.join('common_pid == ' + str(tid) for tid in tids)
            for name in ['sys_enter_write', 'sys_exit_write', 'sys_enter_io_submit', 'sys_exit_io_submit', 'sys_enter_io_getevents', 'sys_exit_io_getevents', 'sys_enter_io_cancel', 'sys_exit_io_cancel', 'sys_enter_io_destroy', 'sys_exit_io_destroy']:
                event = trace / 'events/syscalls' / name
                (event / 'filter').write_text(own_filter + '\n')
                (event / 'enable').write_text('1\n')
            (trace / 'trace').write_text('')
            (trace / 'tracing_on').write_text('1\n')

        def original_entry(target):
            return native_io.original_entry((trace / 'trace').read_text(), target)

        def completed_zero(tid):
            text = (trace / 'trace').read_text()
            lines = text.splitlines()
            for offset, line in enumerate(lines):
                if not re.search(r'\bske-usbd-' + str(tid) + r'\s', line) or 'fd: a, buf: 0, count: 0' not in line:
                    continue
                tail = [value for value in lines[offset + 1:] if re.search(r'\bske-usbd-' + str(tid) + r'\s', value)]
                if tail and re.search(r'(?:sys_exit_write:|sys_write ->)\s*(?:0x0|0)\b', tail[0]):
                    return {'tid': tid, 'fd': 10, 'bytes': 0, 'entry': line, 'exit': tail[0],
                            'actual_trace': text, 'kernel_write_completed_before_any_host_ZLP_read': True}

        cases = [(size, mode) for size in [2047, 2048, 2049, 3071, 3072, 3073]
                 for mode in (['unread', 'partial64', 'zlp'] if size in (2048, 3072) else ['unread', 'partial64'])]
        for index, (total_size, mode) in enumerate(cases):
            label = str(total_size) + '-' + mode
            before = state.reboot.snapshot(f.serial)
            assert count() == index
            payload_size = total_size - 16
            opaque = bytes((i * 17 + total_size) & 255 for i in range(payload_size))
            apdu_response = bytes(Tlv(0x53, opaque)) + b'\x90\x00'
            check(len(apdu_response) + 10 == total_size, 'actual encoded TLV APDU CCID total boundary ' + label)
            piv = PivSession(connection)
            # The stock extended formatter caps an outbound APDU at3062 bytes.
            # Its supported force_short option uses ordinary APDU chaining to
            # provision the near-envelope fixture without changing SDK code.
            if total_size >= 3071:
                piv.protocol = SmartCardProtocol(connection)
                piv.protocol.configure(piv.version, force_short=True)
            piv.authenticate(DEFAULT_MANAGEMENT_KEY)
            history_start = len(connection.history)
            piv.put_object(OBJECT_ID.CHUID, opaque)
            check(piv.get_object(OBJECT_ID.CHUID) == opaque, 'unmodified native PIV SDK stores and reads whole original opaque object ' + label)
            raw_command_history = []
            if total_size >= 3071:
                before_raw = durable()
                command = bytes(Tlv(0x5c, int(OBJECT_ID.CHUID).to_bytes(3, 'big'))) + bytes(Tlv(0x53, opaque))
                apdu = bytes([0, 0xdb, 0x3f, 0xff, 0]) + len(command).to_bytes(2, 'big') + command
                raw_start = len(connection.history)
                value, sw = connection.send_and_receive(apdu)
                raw_command_history = connection.history[raw_start:]
                check(not value and sw == 0x9000 and [h['level'] for h in raw_command_history if h['direction'] == 'out'] == [1, 2],
                      'actual native near-envelope command uses first and final CCID command chain ' + label)
                check(durable() == before_raw and piv.get_object(OBJECT_ID.CHUID) == opaque,
                      'native CCID command chain commits identical object without plaintext or counter changes ' + label)
            row = {'index': index, 'phase': label, 'mode': mode, 'total_response_bytes': total_size,
                   'opaque_object_bytes': payload_size, 'opaque_object_sha256': hashlib.sha256(opaque).hexdigest(),
                   'before_usb': before, 'hotp_counter_before': count(),
                   'stock_sdk_force_short_for_large_provisioning': total_size >= 3071,
                   'native_raw_large_command_history': raw_command_history,
                   'native_before_request_sdk_history': connection.history[history_start:]}
            report['active_cycle'] = row
            saved = durable()
            with state.reboot.connection('FIDO', f.serial) as native_fido:
                management = ManagementSession(native_fido)
                body = Tlv(0x5c, int(OBJECT_ID.CHUID).to_bytes(3, 'big'))
                if mode == 'zlp':
                    control_sequence = connection.write(0x6f, bytes([0, 0xcb, 0x3f, 0xff, len(body)]) + body)
                    control_packet = bytes(d.read(0x82, total_size, timeout=1500))
                    check(struct.unpack('<BIBBBBB', control_packet[:10]) == (0x80, total_size - 10, 0, control_sequence, 0, 0, 0) and control_packet[10:] == apdu_response,
                          'positive control receives original full large maxpacket-multiple response ' + label)
                    check(bytes(d.read(0x82, 1, timeout=1500)) == b'',
                          'positive control receives actual following native zero-length packet ' + label)
                    row['positive_control_sequence'] = control_sequence
                    row['positive_control_full_response_and_native_zlp_verified'] = True
                arm()
                sequence = connection.write(0x6f, bytes([0, 0xcb, 0x3f, 0xff, len(body)]) + body)
                actual_packet_size = min(total_size, CCID_MESSAGE_LENGTH)
                expected = struct.pack('<BIBBBBB', 0x80, actual_packet_size - 10, 0, sequence, 0, 0,
                                       1 if total_size > CCID_MESSAGE_LENGTH else 0) + apdu_response[:actual_packet_size - 10]
                target = wait(lambda: worker(f.bridge.pid, actual_packet_size), 'actual original large native IN write blocked ' + label, 3)
                entry = original_entry(target)
                row.update(original_sequence=sequence, original_packet_bytes=actual_packet_size,
                           expected_original_packet_sha256=hashlib.sha256(expected).hexdigest(), original_worker=target,
                           original_write_entry_before_host_read=entry,
                           original_ccid_response_chain_level=1 if total_size > CCID_MESSAGE_LENGTH else 0)
                check(durable() == saved and count() == index, 'original large GET preserves card plaintext and HOTP ' + label)
                if mode == 'partial64':
                    prefix = bytes(d.read(0x82, 64, timeout=1500))
                    check(prefix == expected[:64], 'actual first native maxpacket equals original header sequence and object prefix ' + label)
                    check(worker(f.bridge.pid, actual_packet_size) == target, 'original large write remains same thread buffer after first64 read ' + label)
                    row['original_prefix_hex'] = prefix.hex()
                    row['host_original_data_bytes_read_before_configuration'] = 64
                elif mode == 'zlp':
                    packet = bytes(d.read(0x82, actual_packet_size, timeout=1500))
                    check(packet == expected, 'entire original large maxpacket-multiple native response read once ' + label)
                    observed = wait(lambda: (lambda value: ('blocked', value) if value else
                                             (lambda completed: ('kernel-buffered', completed) if completed else None)(completed_zero(target['tid'])))
                                    (worker(f.bridge.pid, 0)), 'original large ZLP blocked or completed trace missing ' + label, 3)
                    row['original_data_packet_hex'] = packet.hex()
                    row['original_zero_length_representation'] = observed[0]
                    if observed[0] == 'blocked':
                        zlp_worker = observed[1]
                        row['original_zlp_worker'] = zlp_worker
                        row['original_zlp_write_entry_before_configuration'] = original_entry(zlp_worker)
                        check(zlp_worker['tid'] == target['tid'], 'same original large response worker owns blocked ZLP ' + label)
                    else:
                        row['completed_original_zero_write'] = observed[1]
                        check(observed[1]['tid'] == target['tid'], 'same original large response worker completes unread ZLP before configuration ' + label)
                    row['host_original_data_bytes_read_before_configuration'] = actual_packet_size
                else:
                    row['host_original_data_bytes_read_before_configuration'] = 0
                    check(worker(f.bridge.pid, actual_packet_size) == target, 'whole original large native reply stays unread and blocked ' + label)
                row['durable_plaintext_sha256_at_cutover'] = saved
                start = time.monotonic()
                management.write_device_config(DeviceConfig({TRANSPORT.USB: CAPABILITY(0x202)}), reboot=True)
                row['native_management_ack_seconds'] = time.monotonic() - start
            after = new_generation(before, 0x402)
            row['after_fido_only_usb'] = after
            check(after['classes'] == [3] and after['host_visible_endpoint_addresses'] == [4, 0x84]
                  and after['serial'] == before['serial'], 'actual committed FIDO-only profile after original large reply ' + label)
            check(not (Path('/proc') / str(f.bridge.pid) / 'task' / str(target['tid'])).exists(), 'old large IN worker absent in newly exposed USB generation ' + label)
            if mode == 'zlp' and row['original_zero_length_representation'] == 'blocked':
                check(not (Path('/proc') / str(f.bridge.pid) / 'task' / str(zlp_worker['tid'])).exists(), 'old large ZLP worker absent in new USB generation ' + label)
            row['actual_cutover_write_trace'] = (trace / 'trace').read_text()
            check(durable() == saved and count() == index, 'large or partial transfer cutover has no credential or counter mutation ' + label)
            alive(label + ' cutover')
            try:
                d.read(0x82, CCID_MESSAGE_LENGTH, timeout=300)
            except usb.core.USBError as exc:
                check(exc.errno == 19, 'original partly read or unread CCID handle returns actual ENODEV ' + label)
                row['old_handle_errno'] = exc.errno
            else:
                raise AssertionError('removed original large transfer returned a stale tail ' + label)
            usb.util.dispose_resources(d)
            d = None
            shared.prove_fido(f, credential, label + '-fido-only', report)
            with state.reboot.connection('FIDO', f.serial) as conn:
                check(ManagementSession(conn).read_device_info().config.enabled_capabilities[TRANSPORT.USB] == CAPABILITY(0x202),
                      'stock native Management SDK reads exact committed FIDO-only mask ' + label)
                ManagementSession(conn).write_device_config(DeviceConfig({TRANSPORT.USB: CAPABILITY(0x33a)}), reboot=True)
            restored = new_generation(after, 0x406)
            row['restored_usb'] = restored
            check(restored['host_visible_configuration_descriptor_hex'] == reference['configuration_descriptor_hex'], 'restored profile6 public descriptor bytes ' + label)
            check(durable() == saved and count() == index, 'restored original object and card plaintext retained ' + label)
            alive(label + ' restored')
            d = device(restored)
            connection = NativeLongCcid(d)
            check(bytes(d.read(0x83, 8, timeout=1500)) == b'\x50\x03', 'new occupied slot notification after original large transfer ' + label)
            piv = PivSession(connection)
            actual = piv.get_object(OBJECT_ID.CHUID)
            check(actual == opaque and hashlib.sha256(actual).hexdigest() == row['opaque_object_sha256'], 'fresh native SDK GET returns whole retained object without original tail ZLP or continuation ' + label)
            row['restored_object_sha256'] = hashlib.sha256(actual).hexdigest()
            row['native_restored_object_sdk_history'] = list(connection.history)
            oath = OathSession(connection)
            check({c.id for c in oath.list_credentials()} == {own.id}, 'fresh OATH SDK response contains only original owned credential ' + label)
            value = oath.calculate_code(own).value
            digest = hmac.new(base64.b32decode('JBSWY3DPEHPK3PXP'), index.to_bytes(8, 'big'), 'sha1').digest()
            offset = digest[-1] & 15
            expected_hotp = str((int.from_bytes(digest[offset:offset + 4], 'big') & 0x7fffffff) % 1000000).zfill(6)
            check(value == expected_hotp and count() == index + 1, 'fresh independent HOTP result and one commit after long cutover ' + label)
            shared.prove_fido(f, credential, label + '-restored', report)
            row.update(hotp_counter_after_fresh_sdk=count(), retained_card_plaintext_verified=True,
                       original_services_alive=True, fresh_sdk_no_stale_response_verified=True,
                       old_worker_absent_in_new_generation_verified=True)
            report['cycles'].append(row)
            report.pop('active_cycle')
        alive('final')
        report['final_hotp_counter'] = count()
        report['final_process_status'] = {name: process.poll() for name, process in [('bridge', f.bridge), ('fido', f.process), ('card', f.card.process)]}
        report['native_large_partial_ccid_cutover_verified'] = True
        report['passed'] = True
    except Exception as exc:
        report.update(error=f'{type(exc).__name__}: {exc}', traceback=traceback.format_exc())
        if trace and (trace / 'trace').exists():
            report['last_actual_trace'] = (trace / 'trace').read_text()[-18000:]
    finally:
        if trace:
            for event in ['sys_enter_write', 'sys_exit_write', 'sys_enter_io_submit', 'sys_exit_io_submit', 'sys_enter_io_getevents', 'sys_exit_io_getevents', 'sys_enter_io_cancel', 'sys_exit_io_cancel', 'sys_enter_io_destroy', 'sys_exit_io_destroy']:
                if (trace / 'events/syscalls' / event / 'enable').exists():
                    (trace / 'events/syscalls' / event / 'enable').write_text('0\n')
            if (trace / 'tracing_on').exists():
                (trace / 'tracing_on').write_text('0\n')
        if d is not None:
            usb.util.dispose_resources(d)
        if f:
            for name in ['usb.log', 'daemon.log', 'card/log.txt']:
                p = f.directory / name
                if p.exists(): report[name] = p.read_text(errors='replace')[-18000:]
            try:
                f.stop()
            except Exception as exc:
                report.update(passed=False, cleanup_error=str(exc))
        if tmp and tmp.exists(): shutil.rmtree(tmp)
        report['test_count'] = len(checks)
        for name, key in [('swpasskeyd', 'emulator_binary_sha256'), ('ske-usbd', 'usb_bridge_binary_sha256')]:
            report[key] = hashlib.sha256(Path('/opt', name).read_bytes()).hexdigest()
        report['card_service_sha256'] = hashlib.sha256(Path('/opt/emucard/service.py').read_bytes()).hexdigest()
        report['helper_sha256'] = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
        report['native_io_observer_sha256'] = hashlib.sha256(Path(native_io.__file__).read_bytes()).hexdigest()
        report['shared_helper_sha256'] = hashlib.sha256(Path(shared.__file__).read_bytes()).hexdigest()
        print('SKE_USB_LONG_CUTOVER_BEGIN\n' + json.dumps(report) + '\nSKE_USB_LONG_CUTOVER_END', flush=True)
    return 0 if report['passed'] else 1


if __name__ == '__main__':
    raise SystemExit(main())
